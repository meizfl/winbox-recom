cmake_minimum_required(VERSION 3.16)
project(winbox_lift_probe C)
set(CMAKE_C_STANDARD 11)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(Iconv REQUIRED)
option(WINBOX_STATIC_DEPENDENCIES "Link third-party libraries statically" OFF)
option(WINBOX_FULLY_STATIC "Produce an ELF with no dynamic loader" OFF)
if(WINBOX_STATIC_DEPENDENCIES)
  set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
  set(PKG_CONFIG_ARGN --static)
endif()
if(WINBOX_FULLY_STATIC)
  if(APPLE)
    message(FATAL_ERROR "Apple requires libSystem and system frameworks; use WINBOX_STATIC_DEPENDENCIES")
  endif()
  add_link_options(-static)
endif()
pkg_check_modules(NATIVE_UI REQUIRED IMPORTED_TARGET sdl3 cairo freetype2 fontconfig)
set(WINBOX_INPUT "${CMAKE_CURRENT_SOURCE_DIR}/winbox64.v3.43p10.exe" CACHE FILEPATH "Original PE64 input")
option(WINBOX_FULL_DISCOVERY "Translate all statically discovered instructions (experimental)" ON)
set(LIFT_ARGS)
if(WINBOX_FULL_DISCOVERY)
  list(APPEND LIFT_ARGS --full)
endif()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/gen")
set(LIFT_UNITS)
foreach(N RANGE 0 63)
  list(APPEND LIFT_UNITS "${GEN}/translated_${N}.c")
endforeach()
set(GEN_STAMP "${GEN}/complete.stamp")
add_custom_command(OUTPUT "${GEN_STAMP}"
  BYPRODUCTS "${GEN}/translated.c" "${GEN}/image.c" "${GEN}/imports.c" ${LIFT_UNITS}
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/lift64.py" "${WINBOX_INPUT}" --output "${GEN}" ${LIFT_ARGS}
  COMMAND "${CMAKE_COMMAND}" -E touch "${GEN_STAMP}"
  DEPENDS "${WINBOX_INPUT}" tools/lift64.py tools/pe64.py tools/discover64.py
  VERBATIM)
add_custom_target(winbox_generate
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/update_import_coverage.py" "${GEN}"
  DEPENDS "${GEN_STAMP}" tools/update_import_coverage.py)
add_library(winbox_core STATIC runtime/rt.c runtime/api.c runtime/host.c runtime/crt.c runtime/ui.c runtime/wallpaper_image.c runtime/network.c runtime/threads.c runtime/process.c "${GEN}/translated.c" "${GEN}/image.c" "${GEN}/imports.c" ${LIFT_UNITS})
add_dependencies(winbox_core winbox_generate)
target_include_directories(winbox_core PUBLIC runtime)
if(WINBOX_STATIC_DEPENDENCIES)
  target_compile_definitions(winbox_core PRIVATE WINBOX_PORTABLE_FONTS=1)
endif()
target_link_libraries(winbox_core PUBLIC PkgConfig::NATIVE_UI Threads::Threads Iconv::Iconv m)
add_executable(winbox3 runtime/main.c)
target_link_libraries(winbox3 PRIVATE winbox_core)
add_executable(winbox-probe runtime/main.c)
target_link_libraries(winbox-probe PRIVATE winbox_core)
include(CTest)
if(BUILD_TESTING)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  # Bound compiler RAM/time on the large generated address dispatch tables.
  set_source_files_properties(${LIFT_UNITS} PROPERTIES COMPILE_OPTIONS "-O1")
endif()
